
(function(){
  var all = document.querySelectorAll("div");
  for (var i = 0; i < all.length; i++) {
    var fk = null;
    var ks = Object.keys(all[i]);
    for (var j = 0; j < ks.length; j++) {
      if (ks[j].indexOf("__reactFiber") === 0) {
        fk = ks[j];
        break;
      }
    }
    if (!fk) continue;
    var f = all[i][fk];
    var triesF = 0;
    while (f && triesF < 80) {
      triesF++;
      if (f.memoizedState) {
        var curState = f.memoizedState;
        var hookDepth = 0;
        while (curState && hookDepth < 25) {
          hookDepth++;
          try {
            var q = curState.queue;
            if (q && q.lastRenderedState !== undefined && q.dispatch) {
              var lrs = q.lastRenderedState;
              if (lrs && typeof lrs === "object" && lrs["1"] && lrs["1"].fields) {
                var fields = lrs["1"].fields;
                var validState = {
                  stepValid: lrs["1"].isStepValid,
                  oneLinerValid: fields.one_liner_title ? fields.one_liner_title.isValid : "missing",
                  oneLinerError: fields.one_liner_title ? fields.one_liner_title.errorMessage : "missing",
                  oneLinerDirty: fields.one_liner_title ? fields.one_liner_title.isDirty : "missing",
                  globalError: lrs["1"].globalErrorMessage || "none"
                };
                return JSON.stringify(validState);
              }
            }
          } catch(e) {}
          if (curState.next) curState = curState.next; else break;
        }
      }
      if (f.return) f = f.return; else break;
    }
  }
  return "not found";
})()
