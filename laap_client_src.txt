"""LAAP 客户端抽象 — T04 核心依赖接口。

设计要点：
1. 抽象类 LaapClient 定义星悦三个服务模块需要的最小 LAAP 契约
2. OfflineLaapClient 是纯本地实现（无网络、无 Ollama、无云端 key）
3. 生产环境可替换为 HermesLaapClient，直接 import agi_bridge 调用

关键对齐：
- _compute_deviation 的 0.8/0.5/0.3 三分支逻辑与 agi_bridge.py §Phase 3 完全一致
- sediment_to_memory 产出格式对齐 agi_bridge._render_causal_update 的 Causal Model Update 块
- predict() 在离线模式下记录供 replay() 回填
"""

from __future__ import annotations

import abc
import datetime
import json
import os
import typing

from xingyue_group_agent.models import AttributionResult, CausalPrediction


# ---------------------------------------------------------------------------
# 抽象接口
# ---------------------------------------------------------------------------

class LaapClient(abc.ABC):
    """星悦服务模块消费的 LAAP 能力最小集。"""

    @abc.abstractmethod
    def predict(self, decision_id: str, tool: str, digest: str,
                known_risk: typing.Optional[str] = None) -> CausalPrediction:
        """决策阶段：持久化一条预测记录供 T+7 归因回灌。

        Args:
            decision_id: 本次决策唯一 ID
            tool: 工具名（send_message/post_message/...）
            digest: ≤2000 字决策摘要
            known_risk: 已知风险等级（测试/离线用，跳过因果推理直接指定）
        """

    @abc.abstractmethod
    def replay(self, prediction: CausalPrediction,
               outcome: AttributionResult) -> "ReplayReport":
        """T+7 归因后：将预测+实际结果喂回 LAAP，产出偏差报告。

        内部走 _compute_deviation (纯规则 0.8/0.5/0.3)，
        仅当 deviation ≥ threshold 时写 memory。"""

    @abc.abstractmethod
    def risk_gate(self, tool: str, digest: str,
                  known_risk: typing.Optional[str] = None) -> str:
        """P0-4 防风控闸门：返回 risk_level。

        fail-closed 语义在外层 services.approval_gate 中封装，
        本方法仅返回等级。"""


# ---------------------------------------------------------------------------
# Replay 报告
# ---------------------------------------------------------------------------

class ReplayReport:
    """replay() 的返回结构。"""

    def __init__(
        self,
        decision_id: str,
        predicted_risk: str,
        actual_outcome: str,
        deviation_score: float,
        deviation_band: str,
        action: str,
        sedimented: bool,
    ):
        self.decision_id = decision_id
        self.predicted_risk = predicted_risk
        self.actual_outcome = actual_outcome
        self.deviation_score = deviation_score
        self.deviation_band = deviation_band
        self.action = action
        self.sedimented = sedimented  # True = 入了 memory

    def to_dict(self) -> dict:
        return {
            "decision_id": self.decision_id,
            "predicted_risk": self.predicted_risk,
            "actual_outcome": self.actual_outcome,
            "deviation_score": self.deviation_score,
            "deviation_band": self.deviation_band,
            "action": self.action,
            "sedimented": self.sedimented,
        }

    def __repr__(self) -> str:
        return (
            f"ReplayReport({self.decision_id} "
            f"pred={self.predicted_risk} actual={self.actual_outcome} "
            f"dev={self.deviation_score:.1f}/{self.deviation_band} "
            f"sedimented={self.sedimented})"
        )


# ---------------------------------------------------------------------------
# 离线实现（无网络、无外部依赖）
# ---------------------------------------------------------------------------

# 与 agi_bridge._compute_deviation 完全一致的分支逻辑
_DEVIATION_THRESHOLD = 0.5


def _compute_deviation(predicted_risk: str, actual_success: bool,
                       had_error_signal: bool = False) -> float:
    """纯函数 — 与 LAAP agi_bridge._compute_deviation 同构。"""
    risk = (predicted_risk or "").strip().lower()
    if risk in ("low", "medium") and not actual_success:
        return 0.8   # 低估风险
    if risk in ("high", "critical") and actual_success and not had_error_signal:
        return 0.5   # 高估风险
    return 0.3       # 准确


def _band_from_score(score: float) -> typing.Tuple[str, str]:
    """(band, action) — 与 _render_causal_update 文案对齐。"""
    if score >= 0.8:
        return "underestimated risk", "raise predicted risk level for similar operations"
    if score >= 0.5:
        return "overestimated risk", "may lower predicted risk level for similar benign calls"
    return "accurate", "no model change needed"


def _render_causal_update(
    tool: str,
    predicted_risk: str,
    side_effects: typing.List[str],
    actual_outcome: str,
    score: float,
    band: str,
    action: str,
    tenant_id: str = "",
    ts: typing.Optional[str] = None,
    decision_id: typing.Optional[str] = None,
) -> str:
    """生成 Causal Model Update markdown 块 — 与 agi_bridge._render_causal_update 格式对齐。"""
    ts = ts or datetime.datetime.now(datetime.timezone.utc).isoformat(timespec="seconds")
    se_txt = ", ".join(side_effects) if side_effects else "none"
    tenant_prefix = f"[{tenant_id}] " if tenant_id else ""
    did_line = f"Decision: {decision_id}\n" if decision_id else ""
    return (
        f"\n## Causal Model Update {tenant_prefix}[{ts}]\n"
        f"Tool: {tool}\n"
        f"Predicted: risk={predicted_risk}, side_effects=[{se_txt}]\n"
        f"Actual: {actual_outcome}\n"
        f"{did_line}"
        f"Deviation: {score:.1f} ({band})\n"
        f"Action: {action}\n"
    )


class OfflineLaapClient(LaapClient):
    """纯本地 LAAP 客户端 — 无网络、无 Ollama、无 ingest key。

    用于星悦 T04 离线冒烟 & 单元测试。
    行为与 HermesLaapClient 在输入一致时输出一致（偏差分数、沉淀阈值）。
    """

    def __init__(
        self,
        memory_path: typing.Optional[str] = None,
        deviation_threshold: float = _DEVIATION_THRESHOLD,
    ):
        # 预测暂存（离线用 dict，生产用 Postgres CausalPrediction 表）
        self._predictions: typing.Dict[str, CausalPrediction] = []
        # memory 落盘路径（None = 不落盘，仅跟踪 sedimented 计数）
        self._memory_path = memory_path
        self._threshold = deviation_threshold
        # 观测计数器
        self.predict_count = 0
        self.replay_count = 0
        self.sedimented_count = 0

    # -- 预测存取（SQLite 抽象）-----------------------------------------------

    def predict(self, decision_id: str, tool: str, digest: str,
                known_risk: typing.Optional[str] = None) -> CausalPrediction:
        risk = (known_risk or "medium").strip().lower()
        rec = CausalPrediction(
            decision_id=decision_id,
            risk_level=risk,
            side_effects=[],  # 离线模式不填充（需要真实因果推理）
        )
        self._predictions.append(rec)
        self.predict_count += 1
        return rec

    def get_prediction(self, decision_id: str) -> typing.Optional[CausalPrediction]:
        for rec in self._predictions:
            if rec.decision_id == decision_id:
                return rec
        return None

    # -- 回放 ---------------------------------------------------------------

    def replay(self, outcome: AttributionResult) -> ReplayReport:
        """T+7 回灌入口：按 decision_id 取预测 → 算偏差 → 可能沉淀 memory。"""
        self.replay_count += 1
        pred = self.get_prediction(outcome.decision_id)
        if pred is None:
            raise LookupError(f"CausalPrediction not found: {outcome.decision_id}")

        score = _compute_deviation(
            pred.risk_level, outcome.success,
            had_error_signal=bool(outcome.error_type or outcome.error_message),
        )
        band, action = _band_from_score(score)
        actual_str = "success" if outcome.success else "failure"

        sedimented = False
        if score >= self._threshold:
            block = _render_causal_update(
                tool="send_message",
                predicted_risk=pred.risk_level,
                side_effects=pred.side_effects,
                actual_outcome=actual_str,
                score=score, band=band, action=action,
                tenant_id="",
                decision_id=outcome.decision_id,
            )
            self._sediment(block)
            sedimented = True
            self.sedimented_count += 1

        return ReplayReport(
            decision_id=outcome.decision_id,
            predicted_risk=pred.risk_level,
            actual_outcome=actual_str,
            deviation_score=score,
            deviation_band=band,
            action=action,
            sedimented=sedimented,
        )

    def _sediment(self, block: str) -> None:
        if not self._memory_path:
            return  # 不落盘模式：仅计数
        os.makedirs(os.path.dirname(self._memory_path), exist_ok=True)
        with open(self._memory_path, "a", encoding="utf-8") as fh:
            fh.write(block)

    # -- 闸门 ---------------------------------------------------------------

    def risk_gate(self, tool: str, digest: str,
                  known_risk: typing.Optional[str] = None) -> str:
        """离线闸门：按 known_risk 直接返回；未提供时按关键词简单判断（测试 hook）。"""
        if known_risk:
            return known_risk.strip().lower()
        # 轻量关键词测试 hook：不含已知关键词 → 默认 medium
        lowered = digest.lower()
        if any(w in lowered for w in ("删除", "delete", "覆盖", "overwrite")):
            return "high"
        if any(w in lowered for w in ("block", "屏蔽", "踢出")):
            return "critical"
        return "medium"


# ---------------------------------------------------------------------------
# 便捷：构建一个测试用客户端的工厂
# ---------------------------------------------------------------------------

def build_offline_client(
    memory_path: typing.Optional[str] = None,
    threshold: float = _DEVIATION_THRESHOLD,
) -> OfflineLaapClient:
    return OfflineLaapClient(memory_path=memory_path, deviation_threshold=threshold)
