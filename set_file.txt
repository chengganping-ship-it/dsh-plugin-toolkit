
(function(){
  var fileInput = document.querySelector(\"input[type=file]\");
  if (!fileInput) return \"no file input\";
  // Fetch the image file from the local path
  var xhr = new XMLHttpRequest();
  xhr.open(\"GET\", \"file:///C:/Users/123/.meituan-catpaw/14880026/desk_default_workspace/profile_pic.png\", false);
  xhr.responseType = \"blob\";
  try {
    xhr.send();
    if (xhr.status === 200) {
      var blob = xhr.response;
      var file = new File([blob], \"profile_pic.png\", {type: \"image/png\"});
      var dt = new DataTransfer();
      dt.items.add(file);
      fileInput.files = dt.files;
      fileInput.dispatchEvent(new Event(\"change\", {bubbles: true}));
      fileInput.dispatchEvent(new Event(\"input\", {bubbles: true}));
      return \"file set: \" + fileInput.files.length + \" \" + file.size;
    }
    return \"xhr failed: \" + xhr.status;
  } catch(e) {
    return \"err: \" + e.message;
  }
})()
