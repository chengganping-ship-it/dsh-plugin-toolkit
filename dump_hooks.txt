
(function(){
  var results = [];
  var allElements = document.querySelectorAll("input, button, div, span, a, h1, h2, h3, h4, label");
  for (var i = 0; i < allElements.length; i++) {
    var fk = null;
    var keys = Object.keys(allElements[i]);
    for (var j = 0; j < keys.length; j++) {
      if (keys[j].indexOf("__reactFiber") === 0) {
        fk = keys[j];
        break;
      }
    }
    if (!fk) continue;
    var f = allElements[i][fk];
    var tries = 0;
    while (f && tries < 30) {
      tries++;
      if (f.memoizedState && (tries === 9)) {
        var hookInfo = [];
        var cur = f.memoizedState;
        var depth = 0;
        while (cur && depth < 15) {
          depth++;
          var entry = {d: depth};
          try {
            var ms = cur.memoizedState;
            entry.msType = typeof ms;
            if (typeof ms === "string") entry.msValue = ms;
            else if (typeof ms === "boolean") entry.msValue = ms;
            else if (typeof ms === "number") entry.msValue = ms;
            else if (ms && typeof ms === "object") {
              try { entry.msValue = JSON.stringify(ms).substring(0, 80); } catch(e) { entry.msValue = "[unstringifiable]"; }
            } else entry.msValue = "" + ms;
            if (cur.queue) {
              var lrs = cur.queue.lastRenderedState;
              entry.lrsType = typeof lrs;
              if (typeof lrs === "string") entry.lrsV = lrs;
              else if (typeof lrs === "boolean") entry.lrsV = lrs;
              else if (typeof lrs === "number") entry.lrsV = lrs;
              else if (lrs && typeof lrs === "object") { try { entry.lrsV = JSON.stringify(lrs).substring(0, 80); } catch(e) { entry.lrsV = "[unstring]"; } }
              else entry.lrsV = "" + lrs;
              entry.hasDispatch = !!cur.queue.dispatch;
            }
          } catch(e) { entry.err = e.message; }
          hookInfo.push(entry);
          if (cur.next) cur = cur.next; else break;
        }
        results.push({
          elTag: allElements[i].tagName,
          elClass: (allElements[i].className || "").substring(0, 60),
          fiberType: f.type && (f.type.displayName || f.type.name ? (f.type.displayName || f.type.name) : typeof f.type),
          hooks: hookInfo
        });
      }
      if (f.return) f = f.return; else break;
    }
    if (results.length > 0) break;
  }
  return JSON.stringify(results);
})()
