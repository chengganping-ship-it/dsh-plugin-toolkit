
(function(){
  var inp=document.querySelectorAll("input[type=text]")[1];
  inp.focus();
  var nativeSetter=Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype,"value").set;
  // First clear by selecting all and deleting
  nativeSetter.call(inp, "");
  inp.dispatchEvent(new Event("input",{bubbles:true,cancelable:true}));
  
  function typeChar(el,c){
    var kd=new KeyboardEvent("keydown",{key:c,bubbles:true,cancelable:true});
    var kp=new KeyboardEvent("keypress",{key:c,bubbles:true,cancelable:true});
    var ku=new KeyboardEvent("keyup",{key:c,bubbles:true,cancelable:true});
    el.dispatchEvent(kd);
    el.dispatchEvent(kp);
    var ns=Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype,"value").set;
    ns.call(el, el.value + c);
    var ie=new Event("input",{bubbles:true,cancelable:true});
    el.dispatchEvent(ie);
    el.dispatchEvent(ku);
  }
  var text="ATS Resume Writer";
  for(var i=0;i<text.length;i++){
    typeChar(inp, text[i]);
  }
  var rpKey=Object.keys(inp).find(function(k){return k.startsWith("__reactProps")});
  var rp=inp[rpKey];
  return JSON.stringify({domVal:inp.value,rpVal:rp.value});
})()
