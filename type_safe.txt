
(function(){
  var inp = document.querySelectorAll("input[type=text]")[1];
  // Initialize with a proper typeChar that uses React-compatible events
  function safeTypeChar(el, c) {
    var nativeSetter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, "value").set;
    var beforeVal = el.value;
    var afterVal = beforeVal + c;
    
    // 1. Dispatch keydown
    var kd = new KeyboardEvent("keydown", { key: c, bubbles: true, cancelable: true });
    el.dispatchEvent(kd);
    
    // 2. Dispatch keypress  
    var kp = new KeyboardEvent("keypress", { key: c, charCode: c.charCodeAt(0), bubbles: true, cancelable: true });
    el.dispatchEvent(kp);
    
    // 3. Update value via nativeSetter
    nativeSetter.call(el, afterVal);
    
    // 4. Dispatch input event that React will catch
    var ie = document.createEvent("HTMLEvents");
    ie.initEvent("input", true, true);
    Object.defineProperty(ie, "currentTarget", { writable: false, value: el });
    el.dispatchEvent(ie);
    
    // 5. Dispatch keyup
    var ku = new KeyboardEvent("keyup", { key: c, bubbles: true, cancelable: true });
    el.dispatchEvent(ku);
    
    return { before: beforeVal, after: el.value };
  }
  
  inp.focus();
  var log = [];
  var text = "ATS";
  for (var i = 0; i < text.length; i++) {
    var result = safeTypeChar(inp, text[i]);
    log.push(text[i] + ": " + JSON.stringify(result));
  }
  return JSON.stringify({ log: log, finalValue: inp.value });
})()
