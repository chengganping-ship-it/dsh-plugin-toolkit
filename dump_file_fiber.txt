
(function(){
  var inp = document.querySelector("input[type=file]");
  var fk = null;
  var ks = Object.keys(inp);
  for (var j = 0; j < ks.length; j++) {
    if (ks[j].indexOf("__reactFiber") === 0) {
      fk = ks[j];
      break;
    }
  }
  if (!fk) return "no fiber";
  var f = inp[fk];
  var r = [];
  var t = 0;
  while (f && t < 60) {
    t++;
    if (f.memoizedState) {
      var curState = f.memoizedState;
      var d = 0;
      while (curState && d < 20) {
        d++;
        var ms = curState.memoizedState;
        var lrs = curState.queue ? curState.queue.lastRenderedState : undefined;
        var info = {t:t, d:d, msType:typeof ms, lrsType:typeof lrs};
        if (ms && typeof ms === "object") {
          try { info.msStr = JSON.stringify(ms).substring(0, 150); } catch(e) { info.msStr = "[err]"; }
        } else if (typeof ms === "string" && ms.length > 0) {
          info.msStr = ms.substring(0, 100);
        }
        if (lrs && typeof lrs === "object") {
          try { info.lrsStr = JSON.stringify(lrs).substring(0, 150); } catch(e) { info.lrsStr = "[err]"; }
        } else if (typeof lrs === "string" && lrs.length > 0) {
          info.lrsStr = lrs.substring(0, 100);
        }
        info.hasDispatch = !!(curState.queue && curState.queue.dispatch);
        r.push(info);
        if (curState.next) curState = curState.next; else break;
      }
    }
    if (f.return) f = f.return; else break;
  }
  return JSON.stringify(r.slice(0, 25));
})()
